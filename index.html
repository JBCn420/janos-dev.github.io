⚙️ 1. Time Progression & Tick Engine

The game runs a deterministic 1Hz heartbeat tick loop through a centralized React reducer (gameReducer). It manages active operation timers, automatic state reconciliation, and dynamic offline catch-up calculations upon session initialization. 

// -------------------------------------------------------------------
// 1. CENTRAL SIMULATION TICK LOOP & OFFLINE PROGRESSION CALCULATIONS
// -------------------------------------------------------------------
case 'TICK': {
  const now = Date.now();
  const fields = state.fields.map(f => ({ ...f }));
  let machines = state.machines.map(m => ({ ...m }));
  let silo = [...state.silo];
  let { money, xp, totalHarvestedKg, totalEarned, lastMarketUpdate } = state;

  // Process operation expirations and field status transitions
  const newGrowOps: Operation[] = [];
  const filteredOps = state.operations.filter(op => {
    if (now >= op.startTime + op.durationMs) {
      const field = fields.find(f => f.id === op.fieldId);
      if (!field) return false;

      // Execute machine condition wear upon job completion
      if (op.type !== 'growing') {
        machines.forEach(m => {
          if (op.machineIds.includes(m.id)) {
            m.condition = Math.round(Math.max(0, (m.condition ?? 100) - (1.0 + field.areaHectares * 0.15)) * 10) / 10;
          }
        });
      }

      // State Machine Phase Transitions
      if (op.type === 'plowing') {
        field.status = 'plowed';
        field.assignedMachineIds = [];
        machines.forEach(m => { if (op.machineIds.includes(m.id)) m.assignedFieldId = null; });
      } else if (op.type === 'seeding') {
        field.status = 'growing';
        field.assignedMachineIds = [];
        machines.forEach(m => { if (op.machineIds.includes(m.id)) m.assignedFieldId = null; });
        
        const growMinutes = CROPS[field.crop!]?.growTimeMinutes || 5;
        newGrowOps.push({ 
          id: `grow-${now}-${field.id}`, fieldId: field.id, fieldName: field.name, 
          type: 'growing', machineIds: [], startTime: now, durationMs: growMinutes * 60000, crop: field.crop! 
        });
      } else if (op.type === 'harvesting') {
        field.status = 'idle';
        const baseYieldKg = Math.round(calculateYield(field, op.crop));
        silo = addToSilo(silo, op.crop!, baseYieldKg);
        field.crop = null;
        field.soilQuality = Math.max(20, field.soilQuality - 10);
      }
      return false;
    }
    return true;
  });

  return { ...state, fields, machines, silo, operations: [...filteredOps, ...newGrowOps] };
}pre>

🌱 2. Field Lifecycle & Orphan Operation Recovery

Each field follows a rigid status state machine: idle ➔ plowing ➔ plowed ➔ seeding ➔ growing ➔ ready_to_harvest ➔ harvesting. An automated recovery pass fixes orphaned field states if asynchronous operations lose reference. 

// -------------------------------------------------------------------
// 2. FIELD STATUS TYPE & ORPHAN RECOVERY ALGORITHM
// -------------------------------------------------------------------
export type FieldStatus = 'idle' | 'plowing' | 'plowed' | 'seeding' | 'growing' | 'ready_to_harvest' | 'harvesting';

// Recovery pass executed inside game loop to clean dangling flags:
fields.forEach(f => {
  const hasOp = activeOps.some(op => op.fieldId === f.id);
  if (!hasOp) {
    if (f.status === 'plowing') {
      f.status = 'idle';
      f.assignedMachineIds = [];
    } else if (f.status === 'seeding') {
      f.status = 'plowed';
      f.assignedMachineIds = [];
    } else if (f.status === 'harvesting') {
      f.status = 'ready_to_harvest';
      f.assignedMachineIds = [];
    } else if (f.status === 'growing') {
      if ((f.growthProgress || 0) >= 100) {
        f.status = 'ready_to_harvest';
      }
    }
  }

🚜 3. Machinery Maintenance & Degradation Mathematics

Machinery condition wears down based on the field size (ha). A protective cap limits degradation to a maximum of 15% per single dispatch, protecting game balance on late-game mega estates. 

// -------------------------------------------------------------------
// 3. MAINTENANCE MANAGER FORMULAS & REPAIR CALCULATIONS
// -------------------------------------------------------------------
export class MaintenanceManager {
  public static calculateWear(areaHectares: number, currentCondition: number): number {
    const baseWear = 1.0; // 1.0% base operational wear
    const hectareWear = areaHectares * 0.15; // +0.15% per hectare
    
    let totalWear = baseWear + hectareWear;
    
    // Safety cap: prevent instant destruction on giant estates
    if (totalWear > 15) {
      totalWear = 15;
    }
    
    const newCondition = Math.max(0, currentCondition - totalWear);
    return Math.round(newCondition * 10) / 10;
  }

  public static getRepairCost(machine: Machine, purchasePrice: number): number {
    const lostCondition = 100 - machine.condition;
    if (lostCondition <= 0) return 0;
    // Full restoration (0% to 100%) costs 30% of original asset price
    return Math.round((purchasePrice * 0.3) * (lostCondition / 100));
  }
}pre>

🧮 4. Multi-Factor Crop Yield Calculation

Harvested output is determined dynamically by combining base yield parameters, field soil quality, fertilizer boost status, and regional country suitability bonuses. 

// -------------------------------------------------------------------
// 4. DYNAMIC CROP YIELD COMPUTATION
// -------------------------------------------------------------------
export function calculateYield(field: Field, cropKey?: CropType): number {
  const crop = cropKey || field.crop;
  if (!field || !crop || !CROPS[crop]) return 0;

  const baseYield = CROPS[crop].yieldKgPerHa * field.areaHectares;
  const soilMult = 0.7 + (field.soilQuality / 100) * 0.3;
  const fertMult = field.fertilizerApplied ? 1.25 : 1.0;
  
  const countryData = getCountryData(field.country);
  const countryBonus = countryData.suitableCrops.includes(crop) 
    ? countryData.yieldBonus 
    : 1.0;
  
  return Math.floor(baseYield * soilMult * fertMult * countryBonus);
}pre>
